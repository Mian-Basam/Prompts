</description>

<solution>
// File: components/SearchBar.js
import React, { useState } from "react";

export default function SearchBar() {
  const [searchQuery, setSearchQuery] = useState("");

  return (
    <div className="filter-box">
      <label>Search:</label>
      <input
        value={searchQuery}
        onChange={(e) => setSearchQuery(e.target.value)}
      />
    </div>
  );
}

</solution>

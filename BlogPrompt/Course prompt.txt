<solution>
// File : App.js 
import React, { useState } from "react";
import LeftCounter from "./components/LeftCounter";
import RightCounter from "./components/RightCounter";

export default function App() {
  const [count, setCount] = useState(0);

  const increment = () => setCount((c) => c + 1);

  return (
    <div className="app-container">
      <h2>Exercise 2: Lifted Ownership</h2>
      <LeftCounter count={count} onIncrement={increment} />
      <RightCounter count={count} onIncrement={increment} />
    </div>
  );
}


// File : LeftCounter.js

import React from "react";

export default function LeftCounter({ count, onIncrement }) {
  return (
    <div>
      <p>Left: {count}</p>
      <button onClick={onIncrement}>Increment Left</button>
    </div>
  );
}


// File : RightCounter.js

import React from "react";

  return (
    <div>
      <p>Right: {count}</p>
      <button onClick={onIncrement}>Increment Right</button>
    </div>
  );
}
